cmake_minimum_required(VERSION 3.22)
project(armsx_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_link_options("-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/third_party/SDL" SDL)
# Software scaling, conversion and audio callbacks are real-time work even in
# a debuggable APK. Keep symbols, but avoid shipping SDL's unoptimized Debug loop.
target_compile_options(SDL2 PRIVATE "$<$<CONFIG:Debug>:-O3>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MINIZ_ARCHIVE_APIS ON CACHE BOOL "" FORCE)
set(MINIZ_STDIO ON CACHE BOOL "" FORCE)
add_subdirectory("${REPO}/third_party/libchdr" libchdr)
add_subdirectory("${REPO}/third_party/libadrenotools" adrenotools)
file(GLOB CORE CONFIGURE_DEPENDS "${REPO}/psx/*.c" "${REPO}/psx/dev/*.c" "${REPO}/psx/dev/cdrom/*.c" "${REPO}/psx/input/*.c" "${REPO}/psx/disc/*.c")
file(GLOB RC CONFIGURE_DEPENDS "${REPO}/third_party/rcheevos/src/rcheevos/*.c" "${REPO}/third_party/rcheevos/src/rapi/*.c")
add_library(rcheevos STATIC ${RC}
 "${REPO}/third_party/rcheevos/src/rc_client.c" "${REPO}/third_party/rcheevos/src/rc_compat.c"
 "${REPO}/third_party/rcheevos/src/rc_util.c" "${REPO}/third_party/rcheevos/src/rc_version.c"
 "${REPO}/third_party/rcheevos/src/rhash/cdreader.c" "${REPO}/third_party/rcheevos/src/rhash/hash.c"
 "${REPO}/third_party/rcheevos/src/rhash/hash_disc.c" "${REPO}/third_party/rcheevos/src/rhash/md5.c")
target_include_directories(rcheevos PUBLIC "${REPO}/third_party/rcheevos/include")
target_compile_definitions(rcheevos PUBLIC RC_NO_THREADS=1 RC_HASH_NO_ENCRYPTED RC_HASH_NO_ROM RC_HASH_NO_ZIP)
target_compile_options(rcheevos PRIVATE -O3 -fno-fast-math)
set(FRONTEND argparse.c config.c diagnostics.c toml.c gpu_hw.c gpu_hw_rt.c gpu_hw_gl.c gpu_profile.c host_usage.c perf_hint.c pgo.c archive.cpp main.cpp android_jni.cpp achievements.cpp render.cpp render_sdl.cpp render_gl.cpp render_vk.cpp render_shaders.cpp)
list(TRANSFORM FRONTEND PREPEND "${REPO}/frontend/")
add_library(armsx SHARED ${CORE} ${FRONTEND})
# Identify dirty native sources as well as commits in exported pacing reports.
file(GLOB_RECURSE ARMSX_STAMP_INPUTS CONFIGURE_DEPENDS
    "${REPO}/psx/*.c" "${REPO}/psx/*.h"
    "${REPO}/frontend/*.c" "${REPO}/frontend/*.cpp" "${REPO}/frontend/*.h")
list(APPEND ARMSX_STAMP_INPUTS "${CMAKE_CURRENT_LIST_FILE}")
list(SORT ARMSX_STAMP_INPUTS)
set(ARMSX_STAMP_TEXT "")
foreach(INPUT IN LISTS ARMSX_STAMP_INPUTS)
    file(SHA256 "${INPUT}" DIGEST)
    file(RELATIVE_PATH RELATIVE_INPUT "${REPO}" "${INPUT}")
    string(APPEND ARMSX_STAMP_TEXT "${RELATIVE_INPUT}:${DIGEST}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${INPUT}")
endforeach()
string(SHA256 ARMSX_SOURCE_ID "${ARMSX_STAMP_TEXT}")
string(SUBSTRING "${ARMSX_SOURCE_ID}" 0 16 ARMSX_SOURCE_ID)
target_compile_definitions(armsx PRIVATE ARMSX_SOURCE_ID="${ARMSX_SOURCE_ID}")
option(ARMSX_DIAGNOSTIC_BUILD "Enable native diagnostic probes" OFF)
option(ARMSX_FRAME_TIMING_BUILD "Enable frame and Vulkan wait timing only" OFF)
if(ARMSX_FRAME_TIMING_BUILD)
    target_compile_definitions(armsx PRIVATE ARMSX_FRAME_TIMING_BUILD)
endif()
if(ARMSX_DIAGNOSTIC_BUILD)
    target_compile_definitions(armsx PRIVATE ARMSX_DIAGNOSTIC_BUILD)
endif()
target_include_directories(armsx PRIVATE "${REPO}" "${REPO}/psx" "${REPO}/third_party/libchdr/include" "${REPO}/third_party/libchdr/deps/miniz-3.1.1")
target_compile_definitions(armsx PRIVATE USE_CHD USE_HARDWARE SDL_MAIN_HANDLED __DLL_BUILD OS_INFO=Android ARMSX_ENABLE_GL ARMSX_ENABLE_VULKAN ARMSX_HAVE_ADRENOTOOLS=1)
target_compile_options(armsx PRIVATE -O3 -ffast-math -Wno-address-of-packed-member)
# Allow inlining across the CPU, bus and device-update translation units.
# Keep the per-source strict floating-point overrides below.
target_compile_options(armsx PRIVATE -flto=thin -fno-semantic-interposition)
# The Android app uses one core instance in this library, without replacing
# core functions from another shared object. Bind its internal calls directly.
target_link_options(armsx PRIVATE -flto=thin "-Wl,-Bsymbolic-functions")
set(ARMSX_ANDROID_PGO "off" CACHE STRING "Native PGO: off, generate or use")
set(ARMSX_ANDROID_PGO_PROFILE "" CACHE FILEPATH "Matching LLVM profile for native PGO")
set(ARMSX_ANDROID_PGO_SOURCES ${CORE} ${FRONTEND})
list(REMOVE_ITEM ARMSX_ANDROID_PGO_SOURCES "${REPO}/frontend/pgo.c")
if(NOT ARMSX_ANDROID_PGO STREQUAL "off")
    file(GLOB_RECURSE ARMSX_PGO_HEADERS CONFIGURE_DEPENDS "${REPO}/psx/*.h" "${REPO}/frontend/*.h")
    set(ARMSX_PGO_INPUTS ${CORE} ${FRONTEND} ${ARMSX_PGO_HEADERS} "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
    list(SORT ARMSX_PGO_INPUTS)
    set(ARMSX_PGO_DIGESTS "")
    foreach(ARMSX_PGO_INPUT IN LISTS ARMSX_PGO_INPUTS)
        file(SHA256 "${ARMSX_PGO_INPUT}" ARMSX_PGO_DIGEST)
        file(RELATIVE_PATH ARMSX_PGO_RELATIVE "${REPO}" "${ARMSX_PGO_INPUT}")
        string(APPEND ARMSX_PGO_DIGESTS "${ARMSX_PGO_RELATIVE}:${ARMSX_PGO_DIGEST}\n")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ARMSX_PGO_INPUT}")
    endforeach()
    string(SHA256 ARMSX_PGO_STAMP "${ARMSX_PGO_DIGESTS}")
    set_property(SOURCE "${REPO}/frontend/pgo.c" APPEND PROPERTY COMPILE_DEFINITIONS "ARMSX_PGO_SOURCE_STAMP=\"${ARMSX_PGO_STAMP}\"")
endif()
if(ARMSX_ANDROID_PGO STREQUAL "generate")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/armsx-pgo-source.stamp" "${ARMSX_PGO_STAMP}\n")
    set_property(SOURCE ${ARMSX_ANDROID_PGO_SOURCES} APPEND PROPERTY COMPILE_OPTIONS -fprofile-generate)
    target_link_options(armsx PRIVATE -fprofile-generate)
    set_property(SOURCE "${REPO}/frontend/pgo.c" APPEND PROPERTY COMPILE_DEFINITIONS ARMSX_PGO_GENERATE)
elseif(ARMSX_ANDROID_PGO STREQUAL "use")
    if(NOT EXISTS "${ARMSX_ANDROID_PGO_PROFILE}")
        message(FATAL_ERROR "Native PGO requires an existing profile")
    endif()
    if(NOT EXISTS "${ARMSX_ANDROID_PGO_PROFILE}.stamp")
        message(FATAL_ERROR "Native PGO profile is missing its source stamp")
    endif()
    file(READ "${ARMSX_ANDROID_PGO_PROFILE}.stamp" ARMSX_PGO_PROFILE_STAMP)
    string(STRIP "${ARMSX_PGO_PROFILE_STAMP}" ARMSX_PGO_PROFILE_STAMP)
    if(NOT ARMSX_PGO_PROFILE_STAMP STREQUAL ARMSX_PGO_STAMP)
        message(FATAL_ERROR "Native PGO profile is stale. Retrain it or configure ARMSX_ANDROID_PGO=off.")
    endif()
    set_property(SOURCE "${REPO}/frontend/pgo.c" APPEND PROPERTY COMPILE_DEFINITIONS "ARMSX_PGO_PROFILE_STAMP=\"${ARMSX_PGO_PROFILE_STAMP}\"")
    set_property(SOURCE ${ARMSX_ANDROID_PGO_SOURCES} APPEND PROPERTY COMPILE_OPTIONS "-fprofile-use=${ARMSX_ANDROID_PGO_PROFILE}" -Werror=backend-plugin)
    set_property(SOURCE "${REPO}/frontend/pgo.c" APPEND PROPERTY COMPILE_DEFINITIONS ARMSX_PGO_USE)
elseif(NOT ARMSX_ANDROID_PGO STREQUAL "off")
    message(FATAL_ERROR "Invalid native PGO mode: ${ARMSX_ANDROID_PGO}")
endif()
# Texture coordinates are truncated to integer texel indices. Reciprocal-multiply
# division under fast-math can turn an exact 60 into 59.999996 (BIOS logo quad),
# sampling the preceding atlas row. Keep correctly rounded arithmetic in both
# CPU rasterizers; the rest of the emulator retains its existing optimization.
set_property(SOURCE
    "${REPO}/psx/dev/gpu.c"
    "${REPO}/frontend/gpu_hw_rt.c"
    APPEND PROPERTY COMPILE_OPTIONS "-fno-fast-math"
)
target_link_libraries(armsx PRIVATE rcheevos adrenotools chdr-static miniz SDL2::SDL2 log android GLESv3 EGL OpenSLES m)
add_library(armsx_discord SHARED "${REPO}/android/app/src/main/cpp/discord_bridge.cpp")
target_compile_options(armsx_discord PRIVATE -O2 -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_libraries(armsx_discord PRIVATE log)
set(DISCORD_AAR "${REPO}/android/app/libs/discord_partner_sdk.aar")
set(DISCORD_ENV "${REPO}/android/app/src/main/assets/discord_creds.env")
if(EXISTS "${DISCORD_AAR}" AND EXISTS "${DISCORD_ENV}")
    file(STRINGS "${DISCORD_ENV}" DISCORD_ID_LINE REGEX "^APPLICATION_ID=[0-9]+$")
    if(DISCORD_ID_LINE)
        string(REPLACE "APPLICATION_ID=" "" DISCORD_ID "${DISCORD_ID_LINE}")
        set(DISCORD_SCHEME "discord-${DISCORD_ID}")
        file(STRINGS "${DISCORD_ENV}" DISCORD_SCHEME_LINE REGEX "^APPLICATION_SCHEME=.+")
        if(DISCORD_SCHEME_LINE)
            string(REPLACE "APPLICATION_SCHEME=" "" DISCORD_SCHEME "${DISCORD_SCHEME_LINE}")
            string(REGEX REPLACE "[:/].*$" "" DISCORD_SCHEME "${DISCORD_SCHEME}")
        endif()
        set(DISCORD_STAGE "${CMAKE_CURRENT_BINARY_DIR}/discord-sdk")
        file(ARCHIVE_EXTRACT INPUT "${DISCORD_AAR}" DESTINATION "${DISCORD_STAGE}")
        target_include_directories(armsx_discord PRIVATE "${DISCORD_STAGE}/prefab/modules/discord_partner_sdk/include")
        target_compile_definitions(armsx_discord PRIVATE ARMSX_HAS_DISCORD=1 ARMSX_DISCORD_APPLICATION_ID=${DISCORD_ID} ARMSX_DISCORD_CALLBACK_SCHEME="${DISCORD_SCHEME}")
        target_link_libraries(armsx_discord PRIVATE "${DISCORD_STAGE}/jni/${ANDROID_ABI}/libdiscord_partner_sdk.so")
    endif()
endif()
